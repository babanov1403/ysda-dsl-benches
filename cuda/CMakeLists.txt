cmake_minimum_required(VERSION 3.24)
project(ysda_dsl LANGUAGES CXX CUDA)
find_package(CUDAToolkit 13.3 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_library(ysda_kernels SHARED 00-vector-add/kernel.cu 01-softmax/kernel.cu)
target_link_libraries(ysda_kernels PRIVATE CUDA::cudart)
target_compile_options(ysda_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_executable(cuda_smoke 00-vector-add/main.cu)
target_link_libraries(cuda_smoke PRIVATE ysda_kernels CUDA::cudart)
enable_testing()
add_test(NAME cuda_runtime_smoke COMMAND cuda_smoke)
